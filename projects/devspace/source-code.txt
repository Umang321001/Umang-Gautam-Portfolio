function markdown(s){let out=[],code=false,list=false;for(let line of s.split("\n")){if(line.startsWith("\`\`\`")){if(list){out.push("</ul>");list=false}out.push(code?"</code></pre>":"<pre><code>");code=!code;continue}if(code){out.push(esc(line));continue}if(/^\s*[-*]\s+/.test(line)){if(!list){out.push("<ul>");list=true}out.push("<li>"+inline(line.replace(/^\s*[-*]\s+/,""))+"</li>");continue}else if(list){out.push("</ul>");list=false}if(/^###\s/.test(line))out.push("<h3>"+inline(line.slice(4))+"</h3>");else if(/^##\s/.test(line))out.push("<h2>"+inline(line.slice(3))+"</h2>");else if(/^#\s/.test(line))out.push("<h1>"+inline(line.slice(2))+"</h1>");else if(/^&gt;\s?/.test(line))out.push("<blockquote>"+inline(line.replace(/^&gt;\s?/,""))+"</blockquote>");else if(line.trim())out.push("<p>"+inline(line)+"</p>")}if(list)out.push("</ul>");if(code)out.push("</code></pre>");return out.join("\n")}
